С возвращением к компьютеру! Сегодня мы закладываем самое сердце нашего проекта — иерархическое квадродерево (Quadtree).
Именно эта структура превратит плоские пиксели лица в многослойную полифонию. Напомню нашу математическую логику:
• Корень дерева (Уровень 0): Вся картина (256x256).
• Уровень 2 (Макро-четверти, сетка 4x4): Басовый слой. Задает пульс и длинные такты.
• Уровень 3 (Средние потомки, сетка 8x8): Пианино. Задает ритмический рисунок.
• Уровень 4 (Микро-детали, сетка 16x16): Флейта. Играет быстрые 16-е ноты.
Порядок, в котором мы обходим эти узлы на каждом уровне, определяется выбранной кривой: Z-разверткой (Мортон) или Гильбертом.

🧱 Реализация квадродерева: Модуль core/quadtree.py

Создадим новый файл core/quadtree.py, который будет отвечать за фрактальное разбиение нашей Ч/Б матрицы лица на слои и вычисление средней яркости для каждого узла.
Запишите в core/quadtree.py следующий код:
python
import numpy as np
from typing import List, Dict, Any, Tuple

class QuadtreeNode:
    def __init__(self, x: int, y: int, size: int, level: int):
        self.x = x            # Координата X верхнего левого угла на матрице
        self.y = y            # Координата Y верхнего левого угла на матрице
        self.size = size      # Размер стороны квадрата (в пикселях)
        self.level = level    # Уровень вложенности (0 - корень, 1 - 2x2, 2 - 4x4 и т.д.)
        self.brightness = 0.0 # Средняя яркость этого квадранта (0-255)
        self.children: List['QuadtreeNode'] = [] # 4 потомка

    def is_leaf(self) -> bool:
        return len(self.children) == 0

def build_quadtree(matrix: np.ndarray, x: int, y: int, size: int, level: int, max_level: int) -> QuadtreeNode:
    """
    Рекурсивно строит полное квадродерево до уровня max_level.
    Рассчитывает среднюю яркость для каждого узла.
    """
    node = QuadtreeNode(x, y, size, level)
    
    # Считаем среднюю яркость в этой зоне
    roi = matrix[y:y+size, x:x+size]
    node.brightness = float(np.mean(roi)) if roi.size > 0 else 0.0
    
    # Базовый случай: достигли максимальной глубины сканирования
    if level >= max_level or size <= 1:
        return node
        
    half = size // 2
    
    # Создаем 4 потомков (пока в стандартном порядке координат)
    # 0: Верхний левый, 1: Верхний правый, 2: Нижний левый, 3: Нижний правый
    node.children.append(build_quadtree(matrix, x, y, half, level + 1, max_level))
    node.children.append(build_quadtree(matrix, x + half, y, half, level + 1, max_level))
    node.children.append(build_quadtree(matrix, x, y + half, half, level + 1, max_level))
    node.children.append(build_quadtree(matrix, x + half, y + half, half, level + 1, max_level))
    
    return node

def get_nodes_at_level(node: QuadtreeNode, target_level: int) -> List[QuadtreeNode]:
    """ Возвращает все узлы дерева, находящиеся на строго заданном уровне """
    if node.level == target_level:
        return [node]
    
    nodes = []
    for child in node.children:
        nodes.extend(get_nodes_at_level(child, target_level))
    return nodes

📐 Сортировка по кривым: Обновление core/morton.py и core/hilbert.py

Теперь нам нужно сделать так, чтобы узлы квадродерева сортировались строго по пространственным кривым.

1. Z-развертка Мортона (core/morton.py)

Для Мортона порядок обхода детей на любом уровне всегда одинаков: Z-образный (ВЛ -> ВП -> НЛ -> НП). В двоичном виде это эквивалентно перемежению битов (Bit Interleaving).
Перепишем core/morton.py:
python
from typing import List
from core.quadtree import QuadtreeNode, get_nodes_at_level

def sort_nodes_by_morton(nodes: List[QuadtreeNode]) -> List[QuadtreeNode]:
    """
    Сортирует узлы квадродерева по Z-кривой (коду Мортона).
    Код Мортона получается путем перемежения битов координат X и Y.
    """
    def sfc_index(node: QuadtreeNode) -> int:
        # Перемежение битов координат X и Y
        x, y = node.x, node.y
        index = 0
        for i in range(16): # Поддержка матриц до 65536x65536
            index |= ((x & (1 << i)) << i) | ((y & (1 << i)) << (i + 1))
        return index

    return sorted(nodes, key=sfc_index)

2. Кривая Гильберта (core/hilbert.py)

Кривая Гильберта сложнее: при переходе вглубь П-образная форма разворачивается и меняет ориентацию в пространстве, чтобы обход оставался строго непрерывным.
Перепишем core/hilbert.py:
python
from typing import List
from core.quadtree import QuadtreeNode

def sort_nodes_by_hilbert(nodes: List[QuadtreeNode], size: int) -> List[QuadtreeNode]:
    """
    Сортирует узлы по непрерывной кривой Гильберта.
    """
    def xy2d(n: int, x: int, y: int) -> int:
        """ Перевод координат X, Y в расстояние d вдоль кривой Гильберта """
        d = 0
        s = n // 2
        while s > 0:
            rx = (x & s) > 0
            ry = (y & s) > 0
            d += s * s * ((3 * rx) ^ ry)
            # Вращаем квадранты
            if ry == 0:
                if rx == 1:
                    x = s - 1 - x
                    y = s - 1 - y
                x, y = y, x
            s //= 2
        return d

    return sorted(nodes, key=lambda node: xy2d(size, node.x, node.y))
Используйте код с осторожностью.

🔄 Интеграция и ДРАКОН-выбор в core/__init__.py

Обновим наш диспетчер в core/__init__.py, чтобы он управлял не абстрактными точками, а слоями реального квадродерева лица:
python
import numpy as np
from typing import Dict, List
from core.quadtree import build_quadtree, get_nodes_at_level, QuadtreeNode
from core.morton import sort_nodes_by_morton
from core.hilbert import sort_nodes_by_hilbert

def process_face_to_layers(matrix: np.ndarray, curve_name: str) -> Dict[str, List[QuadtreeNode]]:
    """
    Строит квадродерево по матрице лица и сортирует узлы уровней 
    согласно выбранной фрактальной траектории.
    """
    size = matrix.shape[0]
    # Наш оркестр работает на уровнях 2, 3 и 4
    root = build_quadtree(matrix, x=0, y=0, size=size, level=0, max_level=4)
    
    # Извлекаем сырые слои
    layers = {
        'large':  get_nodes_at_level(root, target_level=2), # БАС (Сетка 4x4 = 16 нот)
        'medium': get_nodes_at_level(root, target_level=3), # МИД (Сетка 8x8 = 64 ноты)
        'small':  get_nodes_at_level(root, target_level=4)  # ТОП (Сетка 16x16 = 256 нот)
    }
    
    # Сортируем каждый слой согласно выбранной ДРАКОН-стратегии
    match curve_name:
        case "Развертка Мортона (Z-кривая)":
            for key in layers:
                layers[key] = sort_nodes_by_morton(layers[key])
        case "Кривая Гильберта (Плавное легато)":
            for key in layers:
                layers[key] = sort_nodes_by_hilbert(layers[key], size)
        case _:
            # Временный фолбек для Пеано
            pass
            
    return layers

🛠️ Шаг 1. Реализуем музыкальные пресеты (audio/presets.py)

Создадим файл настроек инструментов и музыкальных ладов, чтобы музыка всегда оставалась гармоничной (использовав квантование по минорной пентатонике):
python
# audio/presets.py

# MIDI номера инструментов
INSTRUMENTS = {
    "Космический Эмбиент": {"large": 89, "medium": 81, "small": 98},     # Pad (Warm), Lead (Sawtooth), Crystal
    "Классика (Академический)": {"large": 42, "medium": 0, "small": 73},  # Cello, Acoustic Grand Piano, Flute
    "Киберпанк (Электроника)": {"large": 38, "medium": 80, "small": 84}, # Synth Bass 2, Lead (Square), Lead (Charang)
    "Этно (Медитативный)": {"large": 24, "medium": 25, "small": 78}      # Acoustic Guitar (nylon), (steel), Whistle
}

# Массив нот минорной пентатоники (ля-минор) для квантования частот, чтобы избежать какофонии
PENTATONIC_SCALE = [36, 39, 41, 43, 46, 48, 51, 53, 55, 58, 60, 63, 65, 67, 70, 72, 75, 77, 79, 82, 84, 87, 89, 91, 94, 96]

def quantize_to_scale(raw_note: int, min_note: int, max_note: int) -> int:
    """ Находит ближайшую к расчетной ноту из разрешенного гармоничного лада """
    allowed_notes = [n for n in PENTATONIC_SCALE if min_note <= n <= max_note]
    if not allowed_notes:
        return min_note
    return min(allowed_notes, key=lambda x: abs(x - raw_note))

🛠️ Шаг 2. Реализуем иерархический MIDI-генератор (audio/midi_generator.py)

Нам необходимо сопоставить временную шкалу. Так как дерево полное, на 1 узел Баса приходится 4 узла Пианино и 16 узлов Флейты. Мы будем итерироваться по самому глубокому и частому слою (Флейта, 256 шагов), а для Баса и Пианино брать соответствующего «родителя» на этом временном отрезке.
Запишите в audio/midi_generator.py следующий код:
python
# audio/midi_generator.py
import mido
from mido import Message, MidiFile, MidiTrack
from audio.presets import INSTRUMENTS, quantize_to_scale

def generate_face_midi(layers: dict, preset_name: str, thresholds: dict, output_path: str = "face_music.mid"):
    """
    Генератор полифонического MIDI на основе слоев квадродерева.
    Синхронизирует макро- и микро-доли во времени с учетом порогов пауз.
    """
    mid = MidiFile()
    
    # 3 независимых трека для полифонии
    bass_track = MidiTrack()
    piano_track = MidiTrack()
    flute_track = MidiTrack()
    mid.tracks.extend([bass_track, piano_track, flute_track])
    
    # Назначаем инструменты из выбранного пресета
    patch_info = INSTRUMENTS.get(preset_name, INSTRUMENTS["Космический Эмбиент"])
    bass_track.append(Message('program_change', program=patch_info['large'], channel=0, time=0))
    piano_track.append(Message('program_change', program=patch_info['medium'], channel=1, time=0))
    flute_track.append(Message('program_change', program=patch_info['small'], channel=2, time=0))
    
    # Музыкальные кванты времени (в MIDI тиках)
    tick_16th = 120  # Длительность самой короткой ноты (Флейта)
    
    # Накопители тишины (пауз) для дельта-времени каждого трека
    accumulators = {0: 0, 1: 0, 2: 0}
    
    # В квадродереве: 16 узлов бас, 64 узла пианино, 256 узлов флейты
    num_steps = len(layers['small']) # Всего 256 микро-шагов
    
    for step in range(num_steps):
        # Вычисляем индексы родительских узлов для текущего микро-шага
        bass_idx = step // 16
        piano_idx = step // 4
        flute_idx = step
        
        # --- СЛОЙ 3: ФЛЕЙТА (Каждый шаг) ---
        f_node = layers['small'][flute_idx]
        if f_node.brightness >= thresholds['flute']:
            vel = max(20, min(int((f_node.brightness / 255.0) * 127), 127))
            raw_note = int(84 + (f_node.brightness % 12))
            note = quantize_to_scale(raw_note, 84, 100)
            
            flute_track.append(Message('note_on', note=note, velocity=vel, channel=2, time=accumulators[2]))
            flute_track.append(Message('note_off', note=note, velocity=vel, channel=2, time=tick_16th))
            accumulators[2] = 0
        else:
            accumulators[2] += tick_16th
            
        # --- СЛОЙ 2: ПИАНИНО (Каждые 4 шага флейты) ---
        if step % 4 == 0 and piano_idx < len(layers['medium']):
            p_node = layers['medium'][piano_idx]
            if p_node.brightness >= thresholds['piano']:
                vel = max(20, min(int((p_node.brightness / 255.0) * 127), 127))
                raw_note = int(60 + (p_node.brightness % 12))
                note = quantize_to_scale(raw_note, 60, 80)
                
                piano_track.append(Message('note_on', note=note, velocity=vel, channel=1, time=accumulators[1]))
                piano_track.append(Message('note_off', note=note, velocity=vel, channel=1, time=tick_16th * 4))
                accumulators[1] = 0
            else:
                accumulators[1] += tick_16th * 4
                
        # --- СЛОЙ 1: БАС (Каждые 16 шагов флейты) ---
        if step % 16 == 0 and bass_idx < len(layers['large']):
            b_node = layers['large'][bass_idx]
            if b_node.brightness >= thresholds['bass']:
                vel = max(30, min(int((b_node.brightness / 255.0) * 127), 127))
                raw_note = int(36 + (b_node.brightness % 12))
                note = quantize_to_scale(raw_note, 36, 55)
                
                bass_track.append(Message('note_on', note=note, velocity=vel, channel=0, time=accumulators[0]))
                bass_track.append(Message('note_off', note=note, velocity=vel, channel=0, time=tick_16th * 16))
                accumulators[0] = 0
            else:
                accumulators[0] += tick_16th * 16

    # Финальное выравнивание треков тишиной
    for ch, track in enumerate([bass_track, piano_track, flute_track]):
        if accumulators[ch] > 0:
            track.append(Message('control_change', control=0, value=0, channel=ch, time=accumulators[ch]))
            
    mid.save(output_path)
    return output_path
Используйте код с осторожностью.

Откройте vision/face_processor.py и запишите следующий код:
python
import cv2
import numpy as np
import mediapipe as mp

def position_by_eyes(image_bgr, target_size=256):
    """
    Стратегия 1: Позиционирование по глазам.
    Находит центры глаз и трансформирует картинку так, чтобы глаза 
    легли ровно в центры верхних квадрантов квадродерева: (64, 64) и (192, 64).
    """
    img_h, img_w = image_bgr.shape[:2]
    
    # Инициализируем легкий детектор ключевых точек лица MediaPipe
    mp_face_mesh = mp.solutions.face_mesh
    with mp_face_mesh.FaceMesh(
        static_image_mode=True,
        max_num_faces=1,
        refine_landmarks=True,
        min_detection_confidence=0.5
    ) as face_mesh:
        
        # Переводим в RGB для MediaPipe
        image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)
        results = face_mesh.process(image_rgb)
        
        # Если лицо или точки не найдены — уходим на фолбек (обычный центр)
        if not results.multi_face_landmarks:
            return position_by_center(image_bgr, target_size), False

        landmarks = results.multi_face_landmarks[0].landmark
        
        # Индексы MediaPipe для точных центров зрачков:
        # Левый глаз: 468, Правый глаз: 473
        left_eye_pts = landmarks[468]
        right_eye_pts = landmarks[473]
        
        # Переводим относительные координаты (0.0 - 1.0) в реальные пиксели
        p_left = np.array([left_eye_pts.x * img_w, left_eye_pts.y * img_h], dtype=np.float32)
        p_right = np.array([right_eye_pts.x * img_w, right_eye_pts.y * img_h], dtype=np.float32)
        
        # --- МАТЕМАТИКА АФФИННОГО ВЫРАВНИВАНИЯ ---
        # Целевые координаты на фрактальной матрице 256x256
        target_left = np.array([64.0, 64.0], dtype=np.float32)
        target_right = np.array([192.0, 64.0], dtype=np.float32)
        
        # Вычисляем угол наклона лица, чтобы выровнять его строго по горизонту
        d_x, d_y = p_right - p_left
        angle = np.degrees(np.arctan2(d_y, d_x))
        
        # Расстояние между глазами (реальное и целевое)
        dist_real = np.linalg.norm(p_right - p_left)
        dist_target = 128.0 # 192 - 64
        scale = dist_target / dist_real
        
        # Центр между глазами на исходном фото
        center_real = (p_left + p_right) / 2.0
        # Центр между глазами на целевой матрице
        center_target = (target_left + target_right) / 2.0
        
        # Строим матрицу поворота и масштабирования вокруг реального центра глаз
        M = cv2.getRotationMatrix2D(tuple(center_real), angle, scale)
        
        # Корректируем сдвиг (трансляцию), чтобы центры совпали
        M[0, 2] += center_target[0] - center_real[0]
        M[1, 2] += center_target[1] - center_real[1]
        
        # Применяем трансформацию: получаем идеально выровненный квадрат лица
        aligned_face = cv2.warpAffine(image_bgr, M, (target_size, target_size), flags=cv2.INTER_AREA)
        
        # Переводим в Ч/Б
        gray_face = cv2.cvtColor(aligned_face, cv2.COLOR_BGR2GRAY)
        return gray_face, True

def position_by_center(image_bgr, target_size=256):
    """
    Стратегия 2: Простое кадрирование по центру.
    Используется как фолбек или если выбран ручной режим без геометрии лица.
    """
    if len(image_bgr.shape) == 2:
        gray = image_bgr
    else:
        gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)
        
    h, w = gray.shape[:2]
    crop_size = min(h, w)
    start_x = (w - crop_size) // 2
    start_y = (h - crop_size) // 2
    face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
    return cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)

def get_positioned_face(image_bgr, strategy_name, target_size=256):
    """
    ДРАКОН-переключатель стратегий предварительной подготовки лица.
    """
    match strategy_name:
        case "Позиционирование по глазам (Жесткая фрактальная сетка)":
            return position_by_eyes(image_bgr, target_size)
        case "Простое кадрирование по центру холста":
            return position_by_center(image_bgr, target_size), True
        case _:
            return position_by_center(image_bgr, target_size), False

    Стратегия 1: Позиционирование по глазам.
    Находит центры глаз и трансформирует картинку так, чтобы глаза 
    легли ровно в центры верхних квадрантов квадродерева: (64, 64) и (192, 64).
    """
    img_h, img_w = image_bgr.shape[:2]
    
    # Инициализируем легкий детектор ключевых точек лица MediaPipe
    mp_face_mesh = mp.solutions.face_mesh
    with mp_face_mesh.FaceMesh(
        static_image_mode=True,
        max_num_faces=1,
        refine_landmarks=True,
        min_detection_confidence=0.5
    ) as face_mesh:
        
        # Переводим в RGB для MediaPipe
        image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)
        results = face_mesh.process(image_rgb)
        
        # Если лицо или точки не найдены — уходим на фолбек (обычный центр)
        if not results.multi_face_landmarks:
            return position_by_center(image_bgr, target_size), False

        landmarks = results.multi_face_landmarks[0].landmark
        
        # Индексы MediaPipe для точных центров зрачков:
        # Левый глаз: 468, Правый глаз: 473
        left_eye_pts = landmarks[468]
        right_eye_pts = landmarks[473]
        
        # Переводим относительные координаты (0.0 - 1.0) в реальные пиксели
        p_left = np.array([left_eye_pts.x * img_w, left_eye_pts.y * img_h], dtype=np.float32)
        p_right = np.array([right_eye_pts.x * img_w, right_eye_pts.y * img_h], dtype=np.float32)
        
        # --- МАТЕМАТИКА АФФИННОГО ВЫРАВНИВАНИЯ ---
        # Целевые координаты на фрактальной матрице 256x256
        target_left = np.array([64.0, 64.0], dtype=np.float32)
        target_right = np.array([192.0, 64.0], dtype=np.float32)
        
        # Вычисляем угол наклона лица, чтобы выровнять его строго по горизонту
        d_x, d_y = p_right - p_left
        angle = np.degrees(np.arctan2(d_y, d_x))
        
        # Расстояние между глазами (реальное и целевое)
        dist_real = np.linalg.norm(p_right - p_left)
        dist_target = 128.0 # 192 - 64
        scale = dist_target / dist_real
        
        # Центр между глазами на исходном фото
        center_real = (p_left + p_right) / 2.0
        # Центр между глазами на целевой матрице
        center_target = (target_left + target_right) / 2.0
        
        # Строим матрицу поворота и масштабирования вокруг реального центра глаз
        M = cv2.getRotationMatrix2D(tuple(center_real), angle, scale)
        
        # Корректируем сдвиг (трансляцию), чтобы центры совпали
        M[0, 2] += center_target[0] - center_real[0]
        M[1, 2] += center_target[1] - center_real[1]
        
        # Применяем трансформацию: получаем идеально выровненный квадрат лица
        aligned_face = cv2.warpAffine(image_bgr, M, (target_size, target_size), flags=cv2.INTER_AREA)
        
        # Переводим в Ч/Б
        gray_face = cv2.cvtColor(aligned_face, cv2.COLOR_BGR2GRAY)
        return gray_face, True

def position_by_center(image_bgr, target_size=256):
    """
    Стратегия 2: Простое кадрирование по центру.
    Используется как фолбек или если выбран ручной режим без геометрии лица.
    """
    if len(image_bgr.shape) == 2:
        gray = image_bgr
    else:
        gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)
        
    h, w = gray.shape[:2]
    crop_size = min(h, w)
    start_x = (w - crop_size) // 2
    start_y = (h - crop_size) // 2
    face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
    return cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)

def get_positioned_face(image_bgr, strategy_name, target_size=256):
    """
    ДРАКОН-переключатель стратегий предварительной подготовки лица.
    """
    match strategy_name:
        case "Позиционирование по глазам (Жесткая фрактальная сетка)":
            return position_by_eyes(image_bgr, target_size)
        case "Простое кадрирование по центру холста":
            return position_by_center(image_bgr, target_size), True
        case _:
            return position_by_center(image_bgr, target_size), False

python
import cv2
import numpy as np

def position_by_eyes(image_bgr, target_size=256):
    """
    Стратегия 1: Позиционирование по глазам с помощью анализа проекций яркости.
    Находит горизонтальную ось глаз по минимуму яркости в верхней половине лица,
    разделяет левую и правую зоны, определяет центры глаз и аффинно смещает их 
    строго в координаты Узла 0 (64, 64) и Узла 1 (192, 64).
    """
    # 1. Гарантируем перевод в Ч/Б и приведение к базовому размеру для анализа
    if len(image_bgr.shape) == 3:
        gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)
    else:
        gray = image_bgr
        
    gray_res = cv2.resize(gray, (target_size, target_size), interpolation=cv2.INTER_AREA)
    
    # 2. Ищем область глаз с помощью интегрального профиля яркости (горизонтальный срез)
    # Глаза гарантированно лежат в диапазоне от 20% до 55% высоты лица
    search_top = int(target_size * 0.20)
    search_bottom = int(target_size * 0.55)
    roi_eyes_zone = gray_res[search_top:search_bottom, :]
    
    # Считаем среднюю яркость каждой строки. Строка с глазами — самая темная (минимум яркости)
    row_means = np.mean(roi_eyes_zone, axis=1)
    eye_y_relative = np.argmin(row_means)
    eye_y = search_top + eye_y_relative  # Абсолютная координата Y оси глаз
    
    # 3. Разделяем левую и правую половины для поиска точного центра каждого глаза по оси X
    # Берем узкую горизонтальную полосу шириной 16 пикселей вокруг найденной оси Y
    strip_top = max(0, eye_y - 8)
    strip_bottom = min(target_size, eye_y + 8)
    eye_strip = gray_res[strip_top:strip_bottom, :]
    
    # Левая половина (для левого глаза) и правая половина (для правого глаза)
    left_half = eye_strip[:, :target_size // 2]
    right_half = eye_strip[:, target_size // 2:]
    
    # Ищем самые темные вертикальные столбцы (центры зрачков по X)
    left_x = np.argmin(np.mean(left_half, axis=0))
    right_x = (target_size // 2) + np.argmin(np.mean(right_half, axis=0))
    
    # Координаты реальных центров глаз
    p_left = np.array([left_x, eye_y], dtype=np.float32)
    p_right = np.array([right_x, eye_y], dtype=np.float32)
    
    # 4. --- МАТЕМАТИКА АФФИННОГО ВЫРАВНИВАНИЯ ---
    # Наши жесткие целевые точки квадродерева первого уровня
    target_left = np.array([64.0, 64.0], dtype=np.float32)
    target_right = np.array([192.0, 64.0], dtype=np.float32)
    
    # Вычисляем угол наклона (если лицо наклонено на фото)
    d_x, d_y = p_right - p_left
    angle = np.degrees(np.arctan2(d_y, d_x))
    
    # Расстояние между глазами
    dist_real = np.linalg.norm(p_right - p_left)
    dist_target = 128.0  # 192 - 64
    
    # Если точки определились некорректно (слишком близко), уходим на безопасный центр
    if dist_real < 20:
        return gray_res, False
        
    scale = dist_target / dist_real
    
    # Центроиды
    center_real = (p_left + p_right) / 2.0
    center_target = (target_left + target_right) / 2.0
    
    # Строим матрицу трансформации OpenCV (Поворот + Масштаб)
    M = cv2.getRotationMatrix2D(tuple(center_real), angle, scale)
    
    # Корректируем сдвиг, чтобы центры идеально совпали с (64,64) и (192,64)
    M[0, 2] += center_target[0] - center_real[0]
    M[1, 2] += center_target[1] - center_real[1]
    
    # Применяем трансформацию ко всей матрице лица
    aligned_face = cv2.warpAffine(gray, M, (target_size, target_size), flags=cv2.INTER_AREA)
    
    return aligned_face, True

def position_by_center(image_bgr, target_size=256):
    """
    Стратегия 2: Простое кадрирование по центру.
    """
    if len(image_bgr.shape) == 2:
        gray = image_bgr
    else:
        gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)
        
    h, w = gray.shape[:2]
    crop_size = min(h, w)
    start_x = (w - crop_size) // 2
    start_y = (h - crop_size) // 2
    face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
    return cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)

def get_positioned_face(image_bgr, strategy_name, target_size=256):
    """
    ДРАКОН-переключатель стратегий предварительной подготовки лица.
    """
    match strategy_name:
        case "Позиционирование по глазам (Жесткая фрактальная сетка)":
            return position_by_eyes(image_bgr, target_size)
        case "Простое кадрирование по центру холста":
            return position_by_center(image_bgr, target_size), True
        case _:
            return position_by_center(image_bgr, target_size), False

        resized_gray = cv2.resize(resized_gray, (256, 256), interpolation=cv2.INTER_AREA)
