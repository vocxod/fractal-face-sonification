🖼️ Шаг 3. Код каркаса интерактивной панели (main.py)

Создадим скелет нашей панели. На данном этапе программа должна уметь:
1. Загружать изображение лица.
2. Автоматически приводить его к ч/б формату и строгому квадрату (например, 256x256 пикселей), чтобы подготовить матрицу под квадродерево.
3. Иметь боковую панель управления с ручками (выбор кривой, пресеты, пороги яркости).
4. Эмулировать пошаговый «обход» матрицы (пока в виде текстового лога и заглушки).
Запишите следующий код в файл main.py:
python
import streamlit as st
import cv2
import numpy as np
import time

# --- НАСТРОЙКА СТРАНИЦЫ ---
st.set_page_config(
    page_title="Фрактальный аудио-портрет",
    page_icon="🎹",
    layout="wide"
)

st.title("🎹 Фрактальный аудио-портрет: Инструментальный стенд со-творца")
st.caption("Перевод геометрии и яркости лица в многослойную полифоническую музыку")

# --- БОКОВАЯ ПАНЕЛЬ (ИНТЕРФЕЙС СО-ТВОРЦА) ---
st.sidebar.header("🎛️ Панель управления")

# 1. Загрузка фотографии
uploaded_file = st.sidebar.file_uploader("1. Загрузите фото лица", type=["jpg", "jpeg", "png"])

# 2. Выбор математического движка обхода
curve_type = st.sidebar.selectbox(
    "2. Математическая кривая",
    ["Развертка Мортона (Z-кривая)", "Кривая Гильберта (Плавное легато)", "Кривая Пеано (Полиритмия 3x3)"]
)

# 3. Выбор музыкального пака
sound_preset = st.sidebar.selectbox(
    "3. Звуковая палитра (Пресет)",
    ["Космический Эмбиент", "Классика (Академический)", "Киберпанк (Электроника)", "Этно (Медитативный)"]
)

# 4. Настройка порогов яркости (Паузы)
st.sidebar.subheader("🎚️ Пороги активации слоев")
st.sidebar.markdown("<small>Если яркость зоны ниже порога, инструмент берет паузу</small>", unsafe_allowed_html=True)

bass_threshold = st.sidebar.slider("Бас (Крупный слой, L2)", 0, 255, 40)
piano_threshold = st.sidebar.slider("Пианино (Средний слой, L3)", 0, 255, 60)
flute_threshold = st.sidebar.slider("Флейта (Мелкий слой, L4)", 0, 255, 80)


# --- ОСНОВНАЯ ЗОНА ИНТЕРФЕЙСА ---
col_orig, col_fractal = st.columns(2)

if uploaded_file is not None:
    # Чтение изображения через OpenCV
    file_bytes = np.asarray(bytearray(uploaded_file.read()), dtype=np.uint8)
    image = cv2.imdecode(file_bytes, cv2.IMREAD_COLOR)
    
    # Конвертируем в RGB для корректного отображения в Streamlit
    image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
    
    with col_orig:
        st.subheader("🖼️ Оригинал")
        st.image(image_rgb, use_container_width=True)
        
    # --- КАРТИРОВАНИЕ И ПОДГОТОВКА МАТРИЦЫ ---
    # Переводим в градации серого
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    
    # Приводим к строгому квадрату 256x256 (степень двойки 2^8 для Мортона/Гильберта)
    square_size = 256
    resized_gray = cv2.resize(gray, (square_size, square_size), interpolation=cv2.INTER_AREA)
    
    with col_fractal:
        st.subheader("🏁 Фрактальная матрица (256x256)")
        # Отображаем ч/б квадрат, готовый к сканированию
        st.image(resized_gray, width=square_size, channels="GRAY")
        
    # --- ПАНЕЛЬ ВОСПРОИЗВЕДЕНИЯ ---
    st.write("---")
    st.subheader("🎵 Мониторинг проигрывания")
    
    col_btn1, col_btn2, _ = st.columns([1, 1, 4])
    with col_btn1:
        start_play = st.button("▶ ИГРАТЬ", use_container_width=True)
    with col_btn2:
        stop_play = st.button("❚❚ ПАУЗА", use_container_width=True)
        
    if start_play:
        st.success(f"Запущен симулятор обхода по траектории: **{curve_type}**")
        
        # Заглушка под будущий цикл обхода и real-time отрисовку
        status_box = st.empty()
        progress_bar = st.progress(0)
        
        # Имитируем быстрый проход макро-зон (например, 16 шагов L2 уровня)
        for step in range(1, 17):
            status_box.markdown(f"**Шаг {step}/16:** Сканируем сектор. Режим: `{sound_preset}`. "
                                f"Бас: {'🎵 играет' if 120 > bass_threshold else '░ пауза'}")
            progress_bar.progress(step / 16)
            time.sleep(0.2)
            
        st.balloons()

else:
    st.info("💡 Пожалуйста, загрузите фотографию лица в левой панели для начала калибровки звука.")

🛠️ Реализация Шага 2: Модуль vision/face_processor.py

Для поиска лица мы используем классический, быстрый и не требующий тяжелых нейросетевых зависимостей метод — Каскады Хаара (Haar Cascades), который встроен в OpenCV. Он идеально подходит для нашего MVP.
Откройте файл vision/face_processor.py и запишите в него следующий код:
python
import cv2
import numpy as np

def process_and_center_face(image_bgr, target_size=256):
    """
    Находит лицо на фото, вырезает его строгим квадратом,
    центрирует и возвращает ч/б матрицу заданного размера.
    
    image_bgr: исходное изображение, считанное через cv2.imdecode/cv2.imread
    target_size: размер стороны итогового квадрата (256, 512 и т.д.)
    """
    # 1. Переводим в градации серого для работы детектора
    gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)
    
    # 2. Загружаем предобученный каскад Хаара для поиска лиц
    # OpenCV содержит эти файлы внутри своего пакета
    face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
    
    # 3. Ищем лица на фотографии
    faces = face_cascade.detectMultiScale(
        gray,
        scaleFactor=1.1,
        minNeighbors=5,
        minSize=(60, 60) # Игнорируем слишком мелкие шумы
    )
    
    # Если лицо не найдено, берем центральный квадрат всего изображения в качестве фолбека
    if len(faces) == 0:
        h, w = gray.shape[:2]
        crop_size = min(h, w)
        start_x = (w - crop_size) // 2
        start_y = (h - crop_size) // 2
        face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
        resized = cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)
        return resized, False # Возвращаем матрицу и флаг, что лицо найдено не было
        
    # 4. Берем первое найденное лицо (обычно оно самое крупное)
    # x, y — координаты верхнего левого угла; w, h — ширина и высота окна лица
    x, y, w, h = faces[0]
    
    # Нам нужен строгий квадрат. Вычисляем центр найденной зоны лица
    center_x = x + w // 2
    center_y = y + h // 2
    
    # Чтобы портрет был красивым, возьмем рамку чуть шире, чем просто лицо (с запасом на прическу/уши)
    # Коэффициент 1.4 расширяет рамку
    crop_size = int(max(w, h) * 1.4)
    
    # Исходные размеры картинки для контроля границ
    img_h, img_w = gray.shape[:2]
    crop_size = min(crop_size, img_h, img_w) # Рамка не может быть больше самого фото
    
    # Вычисляем новые координаты начала обрезки
    start_x = max(0, center_x - crop_size // 2)
    start_y = max(0, center_y - crop_size // 2)
    
    # Корректируем, если рамка вылетела за правый или нижний край
    if start_x + crop_size > img_w:
        start_x = img_w - crop_size
    if start_y + crop_size > img_h:
        start_y = img_h - crop_size
        
    # 5. Вырезаем квадратную зону лица
    face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
    
    # 6. Масштабируем к нашему целевому фрактальному размеру (например, 256x256)
    resized_face = cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)
    
    return resized_face, True
Используйте код с осторожностью.
