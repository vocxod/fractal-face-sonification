🛠️ Реализация Шага 2: Модуль vision/face_processor.py

Для поиска лица мы используем классический, быстрый и не требующий тяжелых нейросетевых зависимостей метод — Каскады Хаара (Haar Cascades), который встроен в OpenCV. Он идеально подходит для нашего MVP.
Откройте файл vision/face_processor.py и запишите в него следующий код:
python
import cv2
import numpy as np

def process_and_center_face(image_bgr, target_size=256):
    """
    Находит лицо на фото, вырезает его строгим квадратом,
    центрирует и возвращает ч/б матрицу заданного размера.
    
    image_bgr: исходное изображение, считанное через cv2.imdecode/cv2.imread
    target_size: размер стороны итогового квадрата (256, 512 и т.д.)
    """
    # 1. Переводим в градации серого для работы детектора
    gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)
    
    # 2. Загружаем предобученный каскад Хаара для поиска лиц
    # OpenCV содержит эти файлы внутри своего пакета
    face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
    
    # 3. Ищем лица на фотографии
    faces = face_cascade.detectMultiScale(
        gray,
        scaleFactor=1.1,
        minNeighbors=5,
        minSize=(60, 60) # Игнорируем слишком мелкие шумы
    )
    
    # Если лицо не найдено, берем центральный квадрат всего изображения в качестве фолбека
    if len(faces) == 0:
        h, w = gray.shape[:2]
        crop_size = min(h, w)
        start_x = (w - crop_size) // 2
        start_y = (h - crop_size) // 2
        face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
        resized = cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)
        return resized, False # Возвращаем матрицу и флаг, что лицо найдено не было
        
    # 4. Берем первое найденное лицо (обычно оно самое крупное)
    # x, y — координаты верхнего левого угла; w, h — ширина и высота окна лица
    x, y, w, h = faces[0]
    
    # Нам нужен строгий квадрат. Вычисляем центр найденной зоны лица
    center_x = x + w // 2
    center_y = y + h // 2
    
    # Чтобы портрет был красивым, возьмем рамку чуть шире, чем просто лицо (с запасом на прическу/уши)
    # Коэффициент 1.4 расширяет рамку
    crop_size = int(max(w, h) * 1.4)
    
    # Исходные размеры картинки для контроля границ
    img_h, img_w = gray.shape[:2]
    crop_size = min(crop_size, img_h, img_w) # Рамка не может быть больше самого фото
    
    # Вычисляем новые координаты начала обрезки
    start_x = max(0, center_x - crop_size // 2)
    start_y = max(0, center_y - crop_size // 2)
    
    # Корректируем, если рамка вылетела за правый или нижний край
    if start_x + crop_size > img_w:
        start_x = img_w - crop_size
    if start_y + crop_size > img_h:
        start_y = img_h - crop_size
        
    # 5. Вырезаем квадратную зону лица
    face_square = gray[start_y:start_y+crop_size, start_x:start_x+crop_size]
    
    # 6. Масштабируем к нашему целевому фрактальному размеру (например, 256x256)
    resized_face = cv2.resize(face_square, (target_size, target_size), interpolation=cv2.INTER_AREA)
    
    return resized_face, True
Используйте код с осторожностью.

Это отличный настрой перед сном! Проектирование абстракции по правилам ДРАКОН-схем — как раз то, что нужно сделать обстоятельно на чистую голову.
Чтобы пользователь мог в любой момент на лету переключаться между Z-разверткой Мортона (синкопированный ритм), кривой Гильберта (текучее легато) или Пеано, нам нужен единый математический интерфейс. В программировании этот паттерн называется «Стратегия» (Strategy), а на языке ДРАКОН это описывается через блок «Выбор» со строго стандартизированными входами и выходами.
Каждая кривая на входе принимает размер матрицы (например, 256), а на выходе возвращает карту траектории — одномерную последовательность координат (x, y) в том порядке, в котором «звуковой луч» должен обходить пиксели лица.

🛠️ 1. Создаем диспетчер траекторий core/__init__.py

Откройте файл core/__init__.py и запишите туда код, который реализует ДРАКОН-выбор (на базе конструкции match/case из Python 3.11):
python
from typing import List, Tuple
# Импортируем будущие функции генерации траекторий
from core.morton import generate_morton_path
from core.hilbert import generate_hilbert_path
from core.peano import generate_peano_path

def get_fractal_path(curve_name: str, size: int) -> List[Tuple[int, int]]:
    """
    ДРАКОН-переключатель траектории.
    Принимает название кривой из интерфейса и размер квадратной матрицы.
    Возвращает упорядоченный список координат (x, y) для обхода.
    """
    match curve_name:
        case "Развертка Мортона (Z-кривая)":
            return generate_morton_path(size)
            
        case "Кривая Гильберта (Плавное легато)":
            return generate_hilbert_path(size)
            
        case "Кривая Пеано (Полиритмия 3x3)":
            return generate_peano_path(size)
            
        case _:
            # Безопасный фолбек на случай непредвиденного значения
            raise ValueError(f"Неизвестный тип фрактальной кривой: {curve_name}")

🧩 2. Создаем заглушку Z-кривой core/morton.py

Чтобы панель уже завтра могла имитировать реальный обход по пикселям, давайте заложим в core/morton.py математически упрощенный прототип, который мы превратим в полноценное квадродерево на следующем этапе.

python
from typing import List, Tuple

def generate_morton_path(size: int) -> List[Tuple[int, int]]:
    """
    Генерирует траекторию обхода матрицы по Z-кривой Мортона.
    Для старта возвращает базовую сетку, разбитую по квадрантам.
    """
    path = []
    
    # Временный математический прототип: обход макро-пикселей 
    # для демонстрации работы ДРАКОН-переключателя в интерфейсе
    for y in range(size):
        for x in range(size):
            # Будущая логика будет использовать битовое перемежение (Bit Interleaving)
            # или рекурсивное деление квадродерева на 4 потомка
            path.append((x, y))
            
    return path
Используйте код с осторожностью.

🧩 3. Создаем заглушку кривой Гильберта core/hilbert.py

Для симметрии и возможности переключения создадим файл core/hilbert.py:
python
from typing import List, Tuple

def generate_hilbert_path(size: int) -> List[Tuple[int, int]]:
    """
    Генерирует непрерывную траекторию кривой Гильберта (без скачков).
    """
    path = []
    # Заглушка: в полноценной версии здесь будет рекурсивный обход 
    # с поворотом U-образных элементов на 90 градусов
    for x in range(size):
        if x % 2 == 0:
            for y in range(size):
                path.append((x, y))
        else:
            for y in range(size - 1, -1, -1):
                path.append((x, y))
    return path
Используйте код с осторожностью.
